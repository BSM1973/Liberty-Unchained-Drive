cmake_minimum_required(VERSION 3.22)
project(UnchainedDrive VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(formats VST3 Standalone)
if(APPLE)
  list(APPEND formats AU)
endif()
juce_add_plugin(UnchainedDrive COMPANY_NAME "Liberty Music Software" BUNDLE_ID com.libertymusicsoftware.unchaineddrive PLUGIN_MANUFACTURER_CODE Lbty PLUGIN_CODE UcDr PRODUCT_NAME "Unchained Drive" FORMATS ${formats} IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(UnchainedDrive PRIVATE Source/Plugin.cpp Source/Editor.cpp)
target_compile_definitions(UnchainedDrive PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
if(MSVC)
  target_compile_options(UnchainedDrive PRIVATE /utf-8)
endif()
target_link_libraries(UnchainedDrive PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
add_executable(core_test Tests/core_test.cpp)
enable_testing()
add_test(NAME core_test COMMAND core_test)

add_executable(spectral_test Tests/spectral_test.cpp)
target_compile_definitions(spectral_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(spectral_test PRIVATE juce::juce_dsp juce::juce_recommended_config_flags)
add_test(NAME spectral_test COMMAND spectral_test)
